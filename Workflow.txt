# Now create .py file with the code for the strands-agents

# Code for strands-agents

from strands import Agent
from strands.models.ollama import OllamaModel

# Create an Ollama model instance

ollama_model = OllamaModel(host = "http://localhost:11434",  # Ollama server address
                           model_id = "llama3.1:8b")         # Specify which model to use

--> You can also add callback_handler = None as a parameter too to stop automatic replies so 
    that agent replies only when print statement is given as print(agent(user_input)) or if reply = agent(user_input) then print(reply).


# Create an agent using the Ollama model

agent = Agent(model = ollama_model,
              system_prompt = "You are an helpful assistant who chats in a friendly way with the user.")

# Use the agent

while True:
    user_input = input("\nUser:").strip()
    print()

    agent(user_input)
    print()
